    \begin{frame}{Что такое Летняя проектная школа?}
        \begin{outline}
            \1 Командная проектная работа в течение июля по 20 часов в неделю
            \1 Ориентирована на 1-2 курсы
            \1 Проекты от индустрии, кафедр или исследовательских коллективов
                \2 кураторы-практики, \enquote{daily} и ревью кода
                \2 защиты перед комиссией из кураторов
            \1 Научно-развлекательная программа из лекций и мастер-классов%
                \2 приглашаются докладчики из интересных индустриальных и научных проектов
            \1 С 2026 года~--- учебный трек
        \end{outline}
    \end{frame}

    \begin{frame}{Что это даёт?..}
        \begin{block}{\ldots студентам}
        \begin{outline}
            \1 Эквивалент плюс одного семестра практики программирования
            \1 Опыт командной работы и применения индустриальных практик
                \2 git, CI, Scrum-based, разные языки программирования
            \1 Многие проекты многолетние
            \1 Многие проекты вырастают до учебных практик и ВКР
        \end{outline}
        \end{block}

        \pause

        \begin{block}{\ldots преподавателям и партнёрам}
        \begin{outline}
            \1 Отзыв преподавателей из отпуска
            \1 С 2027 года~--- педнагрузка по факультативной практике
                \2 в т.ч. для индустриальных партнёров по постановлению 270 Минцифры
                \2 потребовало изменения календарного графика и введения факультативной практики в чётных семестрах
                    \3 одна РПП с тремя траекториями
            \1 Познакомить студентов со своим стеком и завлечь
        \end{outline}            
        \end{block}
    \end{frame}

    \begin{frame}{Организация школы в целом}
        \begin{outline}
            \1 Подготовка и издание приказа (май)
            \1 Сбор предложений о проектах и лекциях (май~--июнь)
            \1 Составление расписания (конец июня)
                \2 общие мероприятия
                \2 проектные активности
            \1 Трекинг школы (июль)
                \2 сбор статистики, отчёт перед индустриальными партнёрами
                \2 результаты защит, награждения
        \end{outline}

        \pause

        \begin{outline}
            \1 Смешанный формат общих мероприятий и проектных активностей
                \2 очно эффективнее, дистанционно --- популярнее
            \1 Выездные дни к индустриальным партнёрам
            \1 Возможны развлекательно-просветительские мероприятия
        \end{outline}
    \end{frame}

    \begin{frame}{Организация типового проекта}
        \begin{outline}
            \1 Идея (весна)
            \1 Представление проекта (июнь)
            \1 Отбор участников, возможно собеседование или тест (первая неделя после сессии)
            \1 Установочная встреча, внутрипроектные лекции, распределение задач (начало июля)
            \1 Работа в команде (первая половина июля)
            \1 Презентация промежуточных результатов (середина июля)
            \1 Работа в команде (вторая половина июля)
            \1 Финальная презентация (конец июля)
        \end{outline}
    \end{frame}
